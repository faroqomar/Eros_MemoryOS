import AIEvaluationCore
import SwiftUI
import UniformTypeIdentifiers

struct LabJSONDocument: FileDocument {
  static let readableContentTypes: [UTType] = [.json]
  var data: Data
  init(data: Data) { self.data = data }
  init(configuration: ReadConfiguration) throws {
    data = configuration.file.regularFileContents ?? Data()
  }
  func fileWrapper(configuration: WriteConfiguration) throws -> FileWrapper {
    FileWrapper(regularFileWithContents: data)
  }
}

struct LabView: View {
  @Bindable var model: LabViewModel
  let cases: [EvaluationCase]
  @State private var arabic = false
  @State private var exporting = false
  @State private var exportError: String?

  private func text(_ english: String, _ translated: String) -> String {
    arabic ? translated : english
  }

  var body: some View {
    NavigationStack {
      List {
        Section {
          Text(
            text(
              "Fictional data only. This lab does not open your saved memories.",
              "بيانات خيالية فقط. المختبر لا يفتح ذكرياتك المحفوظة.")
          )
          .fixedSize(horizontal: false, vertical: true)
          Text(text("Experiment only. No production choice.", "للتجربة فقط. لم يُعتمد للإنتاج."))
            .font(.subheadline).foregroundStyle(Color.secondary)
            .fixedSize(horizontal: false, vertical: true)
            .accessibilityIdentifier("experiment-notice")
          Toggle(isOn: $arabic) {
            Text("العربية").font(.body).fixedSize(horizontal: false, vertical: true)
          }
        }
        Section {
          ForEach(LabCatalog.candidates) { candidate in
            Button {
              model.select(candidate.id)
            } label: {
              HStack(alignment: .top) {
                VStack(alignment: .leading, spacing: 5) {
                  Text(candidate.title).foregroundStyle(Color.primary)
                    .fixedSize(horizontal: false, vertical: true)
                  Text(candidate.detail).font(.caption).foregroundStyle(Color.secondary)
                    .fixedSize(horizontal: false, vertical: true)
                }
                Spacer(minLength: 4)
                if model.selectedID == candidate.id {
                  Image(systemName: "checkmark.circle.fill").accessibilityHidden(true)
                }
              }.padding(.vertical, 4)
            }
            .disabled(model.running)
            .accessibilityIdentifier(candidate.id)
            .accessibilityAddTraits(model.selectedID == candidate.id ? .isSelected : [])
          }
        } header: {
          Text(text("Configuration", "الإعداد"))
            .font(.headline).fixedSize(horizontal: false, vertical: true)
        }
        Section {
          if model.running {
            VStack(alignment: .leading, spacing: 10) {
              if let progress = model.progress {
                ProgressView(
                  value: Double(progress.completedCases), total: Double(progress.totalCases))
                Text(
                  "\(progress.completedCases) / \(progress.totalCases) · \(progress.currentCaseID)"
                )
                .font(.caption).accessibilityIdentifier("evaluation-progress")
              } else {
                ProgressView(text("Preparing local evaluation…", "جارٍ تجهيز الاختبار المحلي…"))
              }
              Button(text("Cancel", "إلغاء")) { model.cancel() }
            }
          } else {
            Button(text("Run cold + 5 warm suites", "بدء اختبار بارد و٥ اختبارات متكررة")) {
              model.run()
            }
            .accessibilityIdentifier("run-evaluation")
          }
          if let message = model.message { Text(message).foregroundStyle(.orange) }
        }
        Section {
          DisclosureGroup(text("Inspect the 15 shared examples", "عرض الأمثلة المشتركة الخمسة عشر"))
          {
            ForEach(cases) { item in
              VStack(alignment: .leading, spacing: 7) {
                Text("\(item.id) · \(item.language)").font(.headline)
                Text(item.query).textSelection(.enabled)
                Text(item.reviewGuidance).font(.caption).foregroundStyle(.secondary)
              }.padding(.vertical, 6)
            }
          }
          .accessibilityIdentifier("shared-examples")
        }
        if let report = model.report {
          Section(text("Results", "النتائج")) {
            Text("\(report.context.environment.rawValue) · \(report.context.buildConfiguration)")
              .font(.caption).foregroundStyle(.secondary)
            Text(
              text("Source: ", "المصدر: ") + String(report.context.sourceIdentity.prefix(12))
            )
            .font(.caption.monospaced()).textSelection(.enabled)
            .accessibilityIdentifier("source-identity")
            Text(
              report.results.contains(where: { $0.output != nil })
                ? text(
                  "Local model outputs need source and language review. Simulator timing is not phone timing.",
                  "تحتاج إجابات النموذج المحلي إلى مراجعة اللغة والمصادر. وقت المحاكي لا يمثل وقت الهاتف."
                )
                : report.configuration.scope == .retrieval
                  ? text(
                    "Retrieval comparison. This configuration does not request text generation.",
                    "مقارنة استرجاع. هذا الإعداد لا يطلب توليد النص.")
                  : text(
                    "Text generation is unavailable for this run. See the recorded issue below.",
                    "التوليد النصي غير متاح في هذا الاختبار. راجع السبب المسجّل أدناه.")
            )
            .font(.subheadline)
            if let peak = report.telemetry.peakMemoryBytes {
              Text(String(format: "Sampled process footprint: %.1f MiB", Double(peak) / 1_048_576))
                .font(.caption)
            }
            if let bytes = report.telemetry.storageBytes {
              Text(String(format: "Model/index files: %.1f MiB", Double(bytes) / 1_048_576)).font(
                .caption)
            }
            if let repetition = report.context.repetition,
              let count = report.context.repetitionCount
            {
              Text(
                "Suite \(repetition + 1) of up to \(count). Completed reports stay in lab storage."
              ).font(.caption)
            }
            if let indexing = report.indexBuildSeconds {
              Text(String(format: "Index build: %.2f s", indexing)).font(.caption)
            }
            ForEach(report.results) { result in
              VStack(alignment: .leading, spacing: 6) {
                Text("\(result.id) · \(result.language)").font(.headline)
                Text(
                  "Retrieval: \(result.retrievalStatus.rawValue) · Text: \(result.generationStatus.rawValue)"
                )
                .font(.caption).foregroundStyle(.secondary)
                if report.configuration.scope == .text {
                  Text("Full shared corpus supplied; separate retrieval not requested.").font(
                    .caption
                  ).foregroundStyle(.secondary)
                } else {
                  Text(
                    result.retrievedSourceIDs.joined(separator: ", ").isEmpty
                      ? "No sources / لا توجد مصادر"
                      : result.retrievedSourceIDs.joined(separator: ", ")
                  )
                  .font(.caption).textSelection(.enabled)
                }
                if let seconds = result.searchSeconds {
                  Text(String(format: "%.2f ms · retrieval only", seconds * 1000)).font(
                    .caption.monospacedDigit())
                }
                if let exact = result.exactSourceSetMatch {
                  Text(exact ? "Expected source set matched" : "Source set differs from expected")
                    .font(.caption).foregroundStyle(exact ? Color.secondary : Color.orange)
                }
                if let issue = result.retrievalIssue {
                  Text(issue).font(.caption).foregroundStyle(.orange)
                }
                if let issue = result.generationIssue {
                  Text(issue).font(.caption).foregroundStyle(.secondary)
                }
                if let output = result.output {
                  Text(output.text).textSelection(.enabled)
                  if let seconds = result.generationSeconds {
                    Text(
                      String(
                        format: "%.2f s · %@", seconds,
                        output.wasColdLoad == true ? "cold load" : "resident session")
                    )
                    .font(.caption.monospacedDigit())
                  }
                  if let first = output.firstTokenSeconds {
                    Text(String(format: "First token: %.2f s", first)).font(.caption)
                  }
                }
              }.padding(.vertical, 5)
            }
          }
          Section(text("Future phone readiness", "جاهزية الهاتف لاحقًا")) {
            ForEach(report.selectionBlockers, id: \.self) { blocker in Text(blocker).font(.caption)
            }
          }
          Section {
            Button(text("Export JSON report", "تصدير تقرير JSON")) { exporting = true }
              .accessibilityIdentifier("export-report")
          }
        }
        Section(text("Simulator scope", "نطاق المحاكي")) {
          Text(
            text(
              "This task uses the iOS Simulator. The report records available timing, process memory and file bytes. Physical battery use and heat require a later device run.",
              "يستخدم هذا الاختبار محاكي iOS. يسجل التقرير الوقت وذاكرة العملية وحجم الملفات المتاحة. تحتاج البطارية والحرارة إلى اختبار لاحق على جهاز حقيقي."
            )
          )
          .font(.subheadline).foregroundStyle(.secondary)
        }
        if let exportError { Text(exportError).foregroundStyle(.orange) }
      }
      .navigationTitle(text("Local AI Lab", "مختبر الذكاء المحلي"))
      .environment(\.layoutDirection, arabic ? .rightToLeft : .leftToRight)
      .fileExporter(
        isPresented: $exporting, document: LabJSONDocument(data: model.exportData ?? Data()),
        contentType: .json, defaultFilename: "memoryos-ai-evaluation"
      ) { result in
        if case .failure(let error) = result { exportError = error.localizedDescription }
      }
    }
    .onDisappear { model.cancel() }
  }
}
